# Multi-Worktree Fast-Build Pilot Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or
> superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a measured, opt-in Ninja and ccache build path that safely reuses compiler results across Foundry
worktrees while preserving native SCons as the default and fallback.

**Architecture:** `scripts/agent_build.py` remains the user-facing wrapper and selects either native SCons or a
per-configuration Ninja state generated by SCons. ccache is configured per invocation with a shared Foundry cache,
the current worktree as its normalization base, and relative debug paths. A separate stdlib-only benchmark runner
records repeatable resource and timing measurements without mutating source files.

**Tech Stack:** Python 3.8+ stdlib, `unittest`, SCons 4.2+, Ninja 1.13+, ccache 4.x, existing Foundry command-first test
CLI.

---

## Scope and Follow-On Plans

This plan implements rollout steps 1 through 4 of the approved design and produces a usable pilot. Two later plans are
intentionally measurement-gated:

- machine-wide parallelism coordination follows after the pilot compares one full-width build, two half-width builds,
  and four quarter-width builds;
- Linux linker selection and optional Mach-O LLD follow after the benchmark separates compilation, archive, and link
  time.

sccache remains outside the local pilot. The generic `--compiler-cache` boundary introduced here leaves room for a
remote-cache plan when multiple machines become an active requirement.

## File Structure

- Create `scripts/tests/test_agent_build.py`: behavioral unit coverage for wrapper argument resolution, commands,
  cache environments, output identities, and progress records.
- Create `scripts/benchmark_agent_build.py`: safe repeated-command measurement and JSONL result writer.
- Create `scripts/tests/test_benchmark_agent_build.py`: behavioral coverage for benchmark execution and serialization.
- Create `scripts/agent_build.README.md`: local fast-path, fallback, cache, diagnostics, and benchmark instructions.
- Modify `scripts/agent_build.py`: backend selection, per-worktree artifacts, ccache configuration/statistics, Ninja
  state resolution, phase execution, and summaries.
- Modify `SConstruct`: use the current decider spelling and expose a Ninja state-directory option.

No source or binary output directory is shared between worktrees. `.ninja/` remains worktree-local and ignored by
Git. Native SCons continues using `$HOME/.scons_cache`; the Ninja path disables SCons `CacheDir` and uses ccache.

```python
def read_ccache_stats(
    env: dict[str, str],
    *,
    run=subprocess.run,
) -> dict[str, object]:
    completed = run(
        ["ccache", "--print-stats", "--format=json"],
        env=env,
        check=False,
        capture_output=True,
        text=True,
    )
    if completed.returncode != 0:
        return {"error": completed.stderr.strip() or f"ccache exited with {completed.returncode}"}
    try:
        payload = json.loads(completed.stdout)
    except json.JSONDecodeError as exc:
        return {"error": f"invalid ccache statistics JSON: {exc}"}
    return payload if isinstance(payload, dict) else {"error": "ccache statistics were not a JSON object"}


def stats_delta(before: dict[str, object], after: dict[str, object]) -> dict[str, int | float]:
    delta: dict[str, int | float] = {}
    for key, after_value in after.items():
        before_value = before.get(key)
        if isinstance(after_value, (int, float)) and isinstance(before_value, (int, float)):
            delta[key] = after_value - before_value
    return delta


def append_progress_record(path: Path | None, event: str, **fields: object) -> None:
    if path is None:
        return
    path.parent.mkdir(parents=True, exist_ok=True)
    payload = {"version": 1, "event": event, "timestamp": timestamp_utc(), **fields}
    with path.open("a", encoding="utf-8") as progress_file:
        progress_file.write(json.dumps(payload, sort_keys=True) + "\n")
```

```python
#!/usr/bin/env python3
"""Run a command repeatedly and write stdlib-only build benchmark JSONL."""

from __future__ import annotations

import argparse
import json
import resource
import subprocess
import sys
import time
from dataclasses import asdict, dataclass
from pathlib import Path


@dataclass(frozen=True)
class BenchmarkResult:
    version: int
    label: str
    repetition: int
    command: list[str]
    exit_code: int
    wall_ms: int
    user_ms: int
    system_ms: int
    max_rss_kib: int
    input_blocks: int
    output_blocks: int


def run_once(command: list[str], *, label: str, repetition: int) -> BenchmarkResult:
    before = resource.getrusage(resource.RUSAGE_CHILDREN)
    started = time.monotonic()
    completed = subprocess.run(command, check=False)
    wall_ms = int((time.monotonic() - started) * 1000)
    after = resource.getrusage(resource.RUSAGE_CHILDREN)
    return BenchmarkResult(
        version=1,
        label=label,
        repetition=repetition,
        command=command,
        exit_code=completed.returncode,
        wall_ms=wall_ms,
        user_ms=int((after.ru_utime - before.ru_utime) * 1000),
        system_ms=int((after.ru_stime - before.ru_stime) * 1000),
        max_rss_kib=int(after.ru_maxrss),
        input_blocks=int(after.ru_inblock - before.ru_inblock),
        output_blocks=int(after.ru_oublock - before.ru_oublock),
    )


def write_result(output: Path, result: BenchmarkResult) -> None:
    output.parent.mkdir(parents=True, exist_ok=True)
    with output.open("a", encoding="utf-8") as result_file:
        result_file.write(json.dumps(asdict(result), sort_keys=True) + "\n")


def parse_args(argv: list[str]) -> argparse.Namespace:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--label", required=True, help="Stable scenario label written to every record.")
    parser.add_argument("--repeat", type=int, default=3, help="Number of repetitions. Default: 3.")
    parser.add_argument("--output", type=Path, required=True, help="JSONL result path outside tracked fixtures.")
    parser.add_argument("command", nargs=argparse.REMAINDER, help="Command after an optional -- separator.")
    args = parser.parse_args(argv)
    if args.repeat < 1:
        parser.error("--repeat must be at least 1")
    if args.command and args.command[0] == "--":
        args.command = args.command[1:]
    if not args.command:
        parser.error("a benchmark command is required after --")
    return args


def main(argv: list[str]) -> int:
    args = parse_args(argv)
    for repetition in range(1, args.repeat + 1):
        result = run_once(args.command, label=args.label, repetition=repetition)
        write_result(args.output, result)
        print(
            f"[foundry-benchmark] {args.label} {repetition}/{args.repeat}: "
            f"exit={result.exit_code} wall_ms={result.wall_ms}"
        )
        if result.exit_code != 0:
            return result.exit_code
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1:]))
```
